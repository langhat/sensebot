工具介绍:
# 列出目录：
## 调用方式
{
    "name":"ListDir",
    "Path":"<要查看的目录>",
}
# 读取文件：
## 调用方式
{
    "name":"ReadFile",
    "Path":"<要读取的文件路径>",
}
# 写入文件（使用diff方式）：
## 调用方式
{
    "name":"WriteFileDiff",
    "Path":"<要写入的文件路径>",
    "Content":"<要写入的内容>",
    "Mode":"append" 或 "replace"
}
# 文件搜索：
## 调用方式
{
    "name":"SearchFile",
    "Path":"<要搜索的目录>",
    "Pattern":"<搜索模式>",
}
# 文件内容搜索：
## 调用方式
{
    "name":"SearchContent",
    "Path":"<要搜索的文件路径>",
    "Pattern":"<搜索模式>",
}
# 运行 cmd 命令：
## 调用方式
{
    "name":"RunCmd",
    "Command":"<要执行的命令>",
}
# 完成任务
## 调用方式
{
    "name":"TaskDone",
    "msg":"<完成任务后的总结>"
}
